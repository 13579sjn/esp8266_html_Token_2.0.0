void uart_process(void) //回复下发的指令
{
    if(uart2_flag == 0) return;
    uart2_flag = 0;
    re2_buff[256-1] = '\0';
	
    if (strstr((char*)re2_buff, "+MQTTSUBRECV") != NULL)
    {
	esp8266_TIM_flag = 0;
	tim_count_cnt = 0;
        char msg_id[64] = {0};
        char ident[32] = {0};
        char value[32] = {0};
        
        extract_mqtt_info((char*)re2_buff, msg_id, ident, value);
		
        /* ========== 【第一步：立刻回复 OneNet，必须在业务逻辑之前】========== */
        char reply_buf[512] = {0};
        sprintf(reply_buf, 
            "AT+MQTTPUB=0,\"$sys/%s/%s/thing/property/set_reply\",\"{\\\"id\\\":\\\"%s\\\"\\,\\\"code\\\":200\\,\\\"msg\\\":\\\"success\\\"}\",0,0\r\n",
            produce_ID, device_name, msg_id);		
		HAL_Delay(60);
		uart_print(&huart2, reply_buf); //esp8266
		uart_print(&huart1, reply_buf); //调试串口用
		
        /* ========== 【第二步：执行本地业务逻辑】========== */
        if(strstr(value, "true") != NULL || strcmp(value, "1") == 0)
        {
			buzzer_flag = 0;
			UI_LED_State = 2;	
		}
        else if(strstr(value, "false") != NULL || strcmp(value, "0") == 0)
        {
            buzzer_flag = 1;
			helmet = 0;
			people_safe = 0;
			UI_ResetData();
			esp8266_TIM_flag = 0; 
			tim_count_cnt = 0;	
        }
    }
	esp8266_TIM_flag = 1;
    memset(re2_buff, 0, 256);
}
